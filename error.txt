#!/usr/bin/env bash

# ─────────────────────────────────────────
#  sysmonitor – system overview
# ─────────────────────────────────────────

BOLD='\033[1m'
CYAN='\033[0;36m'
GREEN='\033[0;32m'
YELLOW='\033[0;33m'
RESET='\033[0m'

header() {
  echo -e "${CYAN}${BOLD}══════════════════════════════════════${RESET}"
  echo -e "${CYAN}${BOLD}  sysmonitor @ $(hostname)${RESET}"
  echo -e "${CYAN}${BOLD}  $(date '+%Y-%m-%d %H:%M:%S')${RESET}"
  echo -e "${CYAN}${BOLD}══════════════════════════════════════${RESET}"
}

section() {
  echo -e "\n${YELLOW}${BOLD}▸ $1${RESET}"
}

# ── Speichermedien ──────────────────────
section "Speichermedien"
printf "  %-22s %6s %7s %6s %6s  %s\n" "Dateisystem" "Größe" "Benutzt" "Verf." "Verw%" "Eingehängt"
df -h | awk '/^\/dev\// { seen[$1]++; if(seen[$1]==1) printf "  %-22s %6s %7s %6s %6s  %s\n", $1, $2, $3, $4, $5, $6 }'

# ── CPU ─────────────────────────────────
section "CPU"
# Modell
cpu_model=$(grep -m1 "model name" /proc/cpuinfo | cut -d: -f2 | xargs)
cpu_cores=$(nproc)
echo -e "  Modell : $cpu_model"
echo -e "  Kerne  : $cpu_cores"

# Auslastung (1-Sekunden-Snapshot via /proc/stat)
read cpu user nice system idle iowait irq softirq steal _ < /proc/stat
total1=$((user+nice+system+idle+iowait+irq+softirq+steal))
idle1=$idle
sleep 0.5
read cpu user nice system idle iowait irq softirq steal _ < /proc/stat
total2=$((user+nice+system+idle+iowait+irq+softirq+steal))
idle2=$idle
delta_total=$((total2-total1))
delta_idle=$((idle2-idle1))
if [ $delta_total -gt 0 ]; then
  cpu_usage=$(( (delta_total - delta_idle) * 100 / delta_total ))
else
  cpu_usage=0
fi
echo -e "  Last   : ${cpu_usage}%"

# CPU-Temp (beste Zone suchen)
if [ -f /sys/class/thermal/thermal_zone0/temp ]; then
  best_temp=0
  for zone in /sys/class/thermal/thermal_zone*/temp; do
    t=$(cat "$zone" 2>/dev/null)
    [ "$t" -gt "$best_temp" ] 2>/dev/null && best_temp=$t
  done
  [ "$best_temp" -gt 0 ] && echo -e "  Temp   : $((best_temp/1000))°C"
fi

# ── RAM ─────────────────────────────────
section "RAM"
awk '
/^MemTotal:/     { total=$2 }
/^MemAvailable:/ { avail=$2 }
/^SwapTotal:/    { stotal=$2 }
/^SwapFree:/     { sfree=$2 }
END {
  used = total - avail
  printf "  RAM    : %.1f GB benutzt / %.1f GB gesamt\n", used/1024/1024, total/1024/1024
  if (stotal > 0)
    printf "  Swap   : %.1f GB benutzt / %.1f GB gesamt\n", (stotal-sfree)/1024/1024, stotal/1024/1024
}' /proc/meminfo

# ── GPU ─────────────────────────────────
section "GPU"
if command -v nvidia-smi &>/dev/null; then
  nvidia-smi --query-gpu=name,utilization.gpu,memory.used,memory.total,temperature.gpu \
    --format=csv,noheader,nounits | while IFS=',' read name util mem_used mem_total temp; do
    echo -e "  Modell : $(echo $name | xargs)"
    echo -e "  Last   : $(echo $util | xargs)%"
    echo -e "  VRAM   : $(echo $mem_used | xargs) MB / $(echo $mem_total | xargs) MB"
    echo -e "  Temp   : $(echo $temp | xargs)°C"
  done
elif command -v radeontop &>/dev/null; then
  echo -e "  AMD GPU erkannt – starte radeontop für Details"
else
  # Fallback: lspci – nur echte Grafikgeräte (VGA / 3D), kein Sensor Hub
  gpu=$(lspci 2>/dev/null | grep -iE "^[^ ]+ (VGA compatible controller|3D controller|Display controller)")
  if [ -n "$gpu" ]; then
    echo "$gpu" | while read -r line; do
      echo -e "  $line"
    done
  else
    echo -e "  Keine GPU erkannt (kein nvidia-smi / lspci)"
  fi
fi

# ── Netzwerk-Interfaces ─────────────────
section "Netzwerk"
ip -brief addr show 2>/dev/null | awk '{
  printf "  %-12s %-10s %s\n", $1, $2, $3
}'

# ── Uptime ──────────────────────────────
section "System"
uptime_str=$(uptime -p 2>/dev/null || uptime)
echo -e "  Uptime : $uptime_str"
echo -e "  Kernel : $(uname -r)"

echo -e "\n${CYAN}${BOLD}══════════════════════════════════════${RESET}\n"
